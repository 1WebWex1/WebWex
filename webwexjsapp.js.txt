// 1. Анимации появления при прокрутке и прогресс-бар
function initScrollReveal() {
    const revealElements = document.querySelectorAll('.reveal, .reveal-left, .reveal-right, .reveal-scale');
    const observer = new IntersectionObserver((entries) => {
        entries.forEach((entry, index) => {
            if (entry.isIntersecting) {
                setTimeout(() => {
                    entry.target.classList.add('active');
                }, (index % 4) * 80);
                observer.unobserve(entry.target);
            }
        });
    }, { threshold: 0.12 });

    revealElements.forEach(el => observer.observe(el));
}

window.addEventListener('scroll', () => {
    const scrollTop = window.scrollY;
    const docHeight = document.documentElement.scrollHeight - window.innerHeight;
    const scrollPercent = docHeight > 0 ? (scrollTop / docHeight) * 100 : 0;
    const progressEl = document.getElementById('scroll-progress');
    if (progressEl) progressEl.style.width = scrollPercent + '%';

    const navbar = document.getElementById('navbar');
    if (navbar) {
        if (scrollTop > 40) {
            navbar.classList.add('shadow-md', 'bg-white/95');
        } else {
            navbar.classList.remove('shadow-md', 'bg-white/95');
        }
    }
});

// 2. Переключение между Главной страницей и Отдельной страницей Квиза
function switchPage(pageName) {
    const homePage = document.getElementById('page-home');
    const quizPage = document.getElementById('page-quiz');

    if (pageName === 'quiz') {
        homePage.classList.add('hidden');
        quizPage.classList.remove('hidden');
        window.scrollTo({ top: 0, behavior: 'smooth' });
        renderQuizStep();
    } else {
        quizPage.classList.add('hidden');
        homePage.classList.remove('hidden');
        window.scrollTo({ top: 0, behavior: 'smooth' });
        setTimeout(initScrollReveal, 100);
    }
    if (window.lucide) lucide.createIcons();
}

function navigateToSection(sectionId) {
    const homePage = document.getElementById('page-home');
    if (homePage.classList.contains('hidden')) {
        switchPage('home');
        setTimeout(() => {
            const el = document.getElementById(sectionId);
            if (el) el.scrollIntoView({ behavior: 'smooth' });
        }, 150);
    } else {
        const el = document.getElementById(sectionId);
        if (el) el.scrollIntoView({ behavior: 'smooth' });
    }
}

async function submitOrderForm(event) {
    event.preventDefault();
    const nameVal = document.getElementById('lead-name').value.trim();
    const contactVal = document.getElementById('lead-contact').value.trim();
    const commentVal = document.getElementById('lead-comment').value.trim();
    const submitBtn = document.getElementById('lead-submit-btn') || event.target.querySelector('button[type="submit"]');

    const originalBtnText = submitBtn ? submitBtn.textContent : 'Отправить заявку в WebWex';
    if (submitBtn) {
        submitBtn.disabled = true;
        submitBtn.textContent = 'Отправляем заявку...';
    }

    const messageText = [
        '🌿 Новая заявка с сайта WebWex!',
        '',
        `👤 Имя: ${nameVal}`,
        `📞 Контакт: ${contactVal}`,
        `📝 Комментарий / Тариф:\n${commentVal || 'Не указан'}`
    ].join('\n');

    try {
        const response = await fetch(`https://api.telegram.org/bot${TG_BOT_TOKEN}/sendMessage`, {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({
                chat_id: TG_CHAT_ID,
                text: messageText
            })
        });

        if (!response.ok) {
            throw new Error('Telegram API error');
        }

        const summaryBox = document.getElementById('submitted-summary');
        if (commentVal) {
            summaryBox.textContent = `Ваш запрос: ${commentVal}`;
            summaryBox.classList.remove('hidden');
        } else {
            summaryBox.classList.add('hidden');
        }

        document.getElementById('lead-name').value = '';
        document.getElementById('lead-contact').value = '';

        document.getElementById('modal-form-content').classList.add('hidden');
        document.getElementById('modal-success-content').classList.remove('hidden');
        if (window.lucide) lucide.createIcons();
    } catch (error) {
        console.error('Ошибка отправки в Telegram:', error);
        const summaryBox = document.getElementById('submitted-summary');
        summaryBox.textContent = 'Не удалось отправить автоматически. Пожалуйста, напишите нам напрямую в Telegram (@Pchoweb_bot) или ВК.';
        summaryBox.classList.remove('hidden');
        document.getElementById('modal-form-content').classList.add('hidden');
        document.getElementById('modal-success-content').classList.remove('hidden');
        if (window.lucide) lucide.createIcons();
    } finally {
        if (submitBtn) {
            submitBtn.disabled = false;
            submitBtn.textContent = originalBtnText;
        }
    }
}

let currentQuizStep = 0;
let quizAnswers = [];
let calculatedPackageText = "";

function startQuizWithPreset(firstAnswerIndex) {
    quizAnswers = [firstAnswerIndex];
    currentQuizStep = 1;
    switchPage('quiz');
}

function renderQuizStep() {
    const questionArea = document.getElementById('quiz-question-area');
    const resultArea = document.getElementById('quiz-result-area');
    const progressContainer = document.getElementById('quiz-progress-container');

    questionArea.classList.remove('hidden');
    resultArea.classList.add('hidden');
    progressContainer.classList.remove('hidden');

    const q = quizQuestions[currentQuizStep];
    const total = quizQuestions.length;
    const percent = Math.round(((currentQuizStep + 1) / total) * 100);

    document.getElementById('quiz-step-label').textContent = `Вопрос ${currentQuizStep + 1} из ${total}`;
    document.getElementById('quiz-percent-label').textContent = `${percent}%`;
    document.getElementById('quiz-progress-bar').style.width = `${percent}%`;

    document.getElementById('quiz-question-title').textContent = q.title;
    document.getElementById('quiz-question-subtitle').textContent = q.subtitle;

    const grid = document.getElementById('quiz-options-grid');
    grid.innerHTML = '';

    q.options.forEach((opt, idx) => {
        const isSelected = quizAnswers[currentQuizStep] === idx;
        const btn = document.createElement('button');
        btn.type = 'button';
        btn.onclick = () => selectQuizOption(idx);
        btn.className = `text-left p-5 rounded-3xl border transition-all duration-200 flex items-start gap-4 group ${
            isSelected
                ? 'bg-olive text-white border-olive shadow-lg'
                : 'bg-beige-light hover:bg-olive hover:text-white text-olive border-olive/10'
        }`;
        btn.innerHTML = `
            <div class="w-11 h-11 rounded-2xl ${isSelected ? 'bg-terracotta text-white' : 'bg-white text-terracotta group-hover:bg-terracotta group-hover:text-white'} flex items-center justify-center shrink-0 transition-colors">
                <i data-lucide="${opt.icon}" class="w-5 h-5"></i>
            </div>
            <div>
                <div class="font-display font-bold text-base sm:text-lg leading-snug mb-1">${opt.label}</div>
                <div class="text-xs ${isSelected ? 'text-beige/80' : 'text-olive/65 group-hover:text-beige/80'}">${opt.desc}</div>
            </div>
        `;
        grid.appendChild(btn);
    });

    const prevBtn = document.getElementById('quiz-prev-btn');
    if (currentQuizStep > 0) {
        prevBtn.classList.remove('hidden');
    } else {
        prevBtn.classList.add('hidden');
    }

    lucide.createIcons();
}

function selectQuizOption(optionIndex) {
    quizAnswers[currentQuizStep] = optionIndex;
    if (currentQuizStep < quizQuestions.length - 1) {
        currentQuizStep++;
        renderQuizStep();
    } else {
        showQuizResults();
    }
}

function prevQuizStep() {
    if (currentQuizStep > 0) {
        currentQuizStep--;
        renderQuizStep();
    }
}

function resetQuiz() {
    currentQuizStep = 0;
    quizAnswers = [];
    renderQuizStep();
}

// 5. Модальное окно заявки с автозаполнением комментария выбранным тарифом
function openOrderModal(tariffTitle) {
    const modal = document.getElementById('order-modal');
    const commentField = document.getElementById('lead-comment');

    if (tariffTitle) {
        commentField.value = `Интересует тариф: ${tariffTitle}.\nПодарок: + 2 недели бесплатной техподдержки.`;
    } else {
        commentField.value = '';
    }

    document.getElementById('modal-form-content').classList.remove('hidden');
    document.getElementById('modal-success-content').classList.add('hidden');
    modal.classList.remove('hidden');
    modal.classList.add('flex');
    lucide.createIcons();
}

function closeOrderModal() {
    const modal = document.getElementById('order-modal');
    modal.classList.add('hidden');
    modal.classList.remove('flex');
}

// 6. Аккордеон FAQ
function toggleFaq(index) {
    const content = document.getElementById(`faq-content-${index}`);
    const icon = document.getElementById(`faq-icon-${index}`);
    const isHidden = content.classList.contains('hidden');

    if (isHidden) {
        content.classList.remove('hidden');
        icon.style.transform = 'rotate(45deg)';
    } else {
        content.classList.add('hidden');
        icon.style.transform = 'rotate(0deg)';
    }
}
