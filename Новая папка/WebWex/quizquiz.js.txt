/* ============ WebWex · квиз «Какой сайт вам нужен?» ============ */
(() => {
  const $ = s => document.querySelector(s);
  const reduced = matchMedia('(prefers-reduced-motion: reduce)').matches;

  /* Баллы: landing / multi / catalog / blog. Вопрос 7 — выбор платформы. */
  const QUESTIONS = [
    { q: 'Чем вы занимаетесь?', hint: 'Так мы поймём специфику вашей ниши', options: [
      { t: 'Психолог или коуч', e: '🧠', s: { multi: 1, blog: 1 } },
      { t: 'Репетитор, преподаватель', e: '📚', s: { landing: 1, multi: 1 } },
      { t: 'Сантехник, электрик, мастер', e: '🔧', s: { landing: 2 } },
      { t: 'Бьюти-мастер', e: '💅', s: { landing: 1, catalog: 1 } },
      { t: 'Продаю товары / хендмейд', e: '🛍️', s: { catalog: 2 } },
      { t: 'Эксперт, консультант, блогер', e: '✍️', s: { blog: 2 } },
      { t: 'Другое', e: '🌿', s: { multi: 1 } }
    ]},
    { q: 'Какая главная цель сайта?', options: [
      { t: 'Получать заявки на одну услугу', e: '🎯', s: { landing: 3 } },
      { t: 'Рассказать о себе и всех услугах', e: '🏡', s: { multi: 3 } },
      { t: 'Показать товары и работы с ценами', e: '🗂️', s: { catalog: 3 } },
      { t: 'Публиковать статьи и растить доверие', e: '📰', s: { blog: 3 } }
    ]},
    { q: 'Сколько услуг или товаров нужно показать?', options: [
      { t: '1–2, всё на одном экране', e: '☝️', s: { landing: 2 } },
      { t: '3–10, каждой нужна страница', e: '🔢', s: { multi: 2 } },
      { t: 'Больше 10, нужны карточки', e: '🧮', s: { catalog: 2 } },
      { t: 'В основном статьи и материалы', e: '📄', s: { blog: 2 } }
    ]},
    { q: 'Планируете вести блог на сайте?', options: [
      { t: 'Да, буду писать регулярно', e: '🖊️', s: { blog: 3 } },
      { t: 'Иногда: новости и акции', e: '🗓️', s: { multi: 1, blog: 1 } },
      { t: 'Нет, блог не нужен', e: '🙅', s: { landing: 1 } }
    ]},
    { q: 'Нужен ли каталог с фильтрами и карточками?', options: [
      { t: 'Да, обязательно', e: '✅', s: { catalog: 3 } },
      { t: 'Хватит простой галереи работ', e: '🖼️', s: { multi: 1, landing: 1 } },
      { t: 'Нет', e: '❌', s: { landing: 1 } }
    ]},
    { q: 'Какой бюджет на сайт?', hint: 'На Тильде любой тариф на 20% дешевле', options: [
      { t: 'До 10 000 ₽', e: '🌱', s: { landing: 2 } },
      { t: '10 000–20 000 ₽', e: '🌿', s: { landing: 1, multi: 1 } },
      { t: '20 000–30 000 ₽', e: '🌳', s: { multi: 1, catalog: 1 } },
      { t: 'Больше 30 000 ₽', e: '🏔️', s: { multi: 1, catalog: 1, blog: 1 } }
    ]},
    { q: 'На чём соберём ваш сайт?', options: [
      { t: 'Тильда — быстрее и на 20% дешевле', e: '⚡', p: 'tilda' },
      { t: 'Код — полностью уникальный дизайн', e: '💻', p: 'code' },
      { t: 'Не знаю — посоветуйте сами', e: '🤝', p: 'auto' }
    ]}
  ];

  const PLANS = {
    landing: { name: 'Лендинг «Росток»', emoji: '🌱', price: 9999, tilda: 7999, from: true,
      why: 'Одна услуга — одна страница, которая ведёт посетителя к заявке. Идеально, чтобы быстро запуститься и начать получать клиентов.' },
    multi: { name: 'Многостраничный сайт «Лиана»', emoji: '🌿', price: 18999, tilda: 15199, from: true,
      why: 'Несколько направлений заслуживают отдельных страниц: клиентам проще найти нужное, а поисковикам — приводить вам трафик.' },
    catalog: { name: 'Сайт-каталог «Крона»', emoji: '🌳', price: 21999, tilda: 17599, from: false,
      why: 'Карточки, фильтры и цены покажут весь ассортимент лицом — клиент выберет и закажет без лишних вопросов.' },
    blog: { name: 'Сайт-блог «Баобаб»', emoji: '🌴', price: 23999, tilda: 19199, from: true,
      why: 'Регулярные статьи растут в поиске и прогревают аудиторию: читатели постепенно становятся клиентами.' }
  };
  const ORDER = ['landing', 'multi', 'catalog', 'blog'];

  let step = 0, platform = 'auto', answers = [];
  let scores = { landing: 0, multi: 0, catalog: 0, blog: 0 };

  const body = $('#qBody'), fill = $('#qFill'), label = $('#qLabel'),
        back = $('#qBack'), quizBox = $('#quizBox'), resultBox = $('#resultBox');

  function render() {
    const q = QUESTIONS[step];
    fill.style.width = (step / QUESTIONS.length * 100) + '%';
    label.textContent = `Вопрос ${step + 1} из ${QUESTIONS.length}`;
    back.disabled = step === 0;
    body.innerHTML = `
      <h2 class="q-title">${q.q}</h2>
      ${q.hint ? `<p class="q-hint">${q.hint}</p>` : ''}
      <div class="q-options">
        ${q.options.map((o, i) => `<button class="option" data-i="${i}"><span class="emoji">${o.e}</span><span>${o.t}</span></button>`).join('')}
      </div>`;
    body.querySelectorAll('.option').forEach(btn =>
      btn.addEventListener('click', () => pick(+btn.dataset.i, btn))
    );
  }

  function pick(i, btn) {
    if (btn.classList.contains('picked')) return;
    btn.classList.add('picked');
    const o = QUESTIONS[step].options[i];
    answers.push({ s: o.s || null, p: o.p || null, label: o.t });
    if (o.s) for (const k in o.s) scores[k] += o.s[k];
    if (o.p) platform = o.p;
    setTimeout(() => { step++; swap(step < QUESTIONS.length ? render : showResult); }, 230);
  }

  function swap(fn) {
    body.classList.add('q-out');
    setTimeout(() => {
      fn();
      body.classList.remove('q-out');
      body.classList.add('q-in');
      setTimeout(() => body.classList.remove('q-in'), 400);
    }, 220);
  }

  back.addEventListener('click', () => {
    if (step === 0) return;
    const a = answers.pop();
    if (a.s) for (const k in a.s) scores[k] -= a.s[k];
    if (a.p) platform = 'auto';
    step--;
    swap(render);
  });

  const fmt = n => n.toLocaleString('ru-RU');

  function showResult() {
    let key = ORDER[0];
    for (const k of ORDER) if (scores[k] > scores[key]) key = k;
    const p = PLANS[key];
    const bestBox = platform === 'code' ? 'code' : 'tilda';

    quizBox.hidden = true;
    resultBox.hidden = false;
    $('#resEmoji').textContent = p.emoji;
    $('#resName').textContent = p.name;
    $('#resWhy').textContent = p.why;
    $('#resPriceCode').textContent = (p.from ? 'от ' : '') + fmt(p.price) + ' ₽';
    $('#resPriceTilda').textContent = (p.from ? 'от ' : '') + fmt(p.tilda) + ' ₽';
    $('#boxCode').classList.toggle('best', bestBox === 'code');
    $('#boxTilda').classList.toggle('best', bestBox === 'tilda');
    $('#flagCode').hidden = bestBox !== 'code';
    $('#flagTilda').hidden = bestBox !== 'tilda';
    $('#flagCode').textContent = 'ваш выбор';
    $('#flagTilda').textContent = platform === 'tilda' ? 'ваш выбор' : 'выгода −20%';
    $('#resChips').innerHTML = answers.map(a => `<span>${a.label}</span>`).join('');
    $('#resOrder').href = `../index.html?plan=${key}&platform=${platform}#contacts`;
    confetti();
  }

  $('#resRestart').addEventListener('click', () => {
    step = 0; platform = 'auto'; answers = [];
    scores = { landing: 0, multi: 0, catalog: 0, blog: 0 };
    resultBox.hidden = true;
    quizBox.hidden = false;
    swap(render);
  });

  /* Листопад-конфетти */
  function confetti() {
    if (reduced) return;
    const colors = ['#a8bd57', '#c9d97f', '#7c8f3e'];
    for (let i = 0; i < 26; i++) {
      const s = document.createElement('span');
      s.className = 'q-leaf';
      s.style.left = Math.random() * 100 + '%';
      s.style.background = colors[i % 3];
      s.style.setProperty('--t', (2.4 + Math.random() * 2) + 's');
      s.style.setProperty('--d', (Math.random() * .9) + 's');
      s.style.setProperty('--r', (180 + Math.random() * 360) + 'deg');
      s.addEventListener('animationend', () => s.remove());
      resultBox.appendChild(s);
    }
  }

  /* Падающие листья на фоне */
  const jungle = $('#jungle');
  if (jungle && !reduced) {
    let live = 0;
    setInterval(() => {
      if (document.hidden || live >= 8) return;
      live++;
      const l = document.createElement('span');
      l.className = 'fall-leaf';
      const size = 10 + Math.random() * 8;
      l.style.width = l.style.height = size + 'px';
      l.style.left = Math.random() * 100 + 'vw';
      l.style.background = 'linear-gradient(135deg, #a8bd57, #55662c)';
      l.style.setProperty('--t', (9 + Math.random() * 6) + 's');
      l.style.setProperty('--dx', (Math.random() * 140 - 70) + 'px');
      l.style.setProperty('--r', (240 + Math.random() * 300) + 'deg');
      l.addEventListener('animationend', () => { l.remove(); live--; });
      jungle.appendChild(l);
    }, 2200);
  }

  render();
})();
